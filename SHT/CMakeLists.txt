set(SFXS fly omp kernel)
set(SH_to_spat_COMPONENTS ST S T Q QST)
set(spat_to_SH_COMPONENTS Q ST QST)

function(COMP_TO_FILENAME_FRAGMENT COMP FRAG)
	if(COMP STREQUAL Q)
		set(${FRAG} "" PARENT_SCOPE)
	else()
		string(TOLOWER ${COMP} TEMP_FRAG)
		set(${FRAG} ${TEMP_FRAG} PARENT_SCOPE)
	endif()
endfunction()

set(CODELET_OUTPUTS)
foreach(SFX IN LISTS SFXS)
	foreach(COMPONENT IN LISTS SH_to_spat_COMPONENTS)
		comp_to_filename_fragment(${COMPONENT} FRAG)
		set(INPUT_FILE "${CMAKE_CURRENT_LIST_DIR}/${SFX}_SH_to_spat.gen.c")
		set(OUTPUT_FILE "${CMAKE_CURRENT_BINARY_DIR}/SH${FRAG}_to_spat_${SFX}.c")
		list(APPEND CODELET_OUTPUTS "${OUTPUT_FILE}")
		string(REPLACE "${PROJECT_BINARY_DIR}/" "" OUTPUT_RELATIVE_PATH "${OUTPUT_FILE}")

		add_custom_command(
			OUTPUT "${OUTPUT_FILE}"
			COMMAND ${CMAKE_COMMAND}
			ARGS -D INPUT_FILE=${INPUT_FILE}
					 -D OUTPUT_FILE=${OUTPUT_FILE}
					 -D COMPONENT=${COMPONENT}
					 -D PARITY=EO
					 -P ${PROJECT_SOURCE_DIR}/cmake/Codelet.cmake
			DEPENDS ${INPUT_FILE}
			COMMENT "Generating C codelet ${OUTPUT_RELATIVE_PATH}"
		)
	endforeach()

	foreach(COMPONENT IN LISTS spat_to_SH_COMPONENTS)
		comp_to_filename_fragment(${COMPONENT} FRAG)
		set(INPUT_FILE "${CMAKE_CURRENT_LIST_DIR}/${SFX}_spat_to_SH.gen.c")
		set(OUTPUT_FILE "${CMAKE_CURRENT_BINARY_DIR}/spat_to_SH${FRAG}_${SFX}.c")
		list(APPEND CODELET_OUTPUTS "${OUTPUT_FILE}")
		string(REPLACE "${PROJECT_BINARY_DIR}/" "" OUTPUT_RELATIVE_PATH "${OUTPUT_FILE}")

		add_custom_command(
			OUTPUT "${OUTPUT_FILE}"
			COMMAND ${CMAKE_COMMAND}
			ARGS -D INPUT_FILE=${INPUT_FILE}
					 -D OUTPUT_FILE=${OUTPUT_FILE}
					 -D COMPONENT=${COMPONENT}
					 -D PARITY=EO
					 -P ${PROJECT_SOURCE_DIR}/cmake/Codelet.cmake
			DEPENDS ${INPUT_FILE}
			COMMENT "Generating C codelet ${OUTPUT_RELATIVE_PATH}"
		)
	endforeach()
endforeach()

add_custom_command(
	OUTPUT cuda_legendre.inc
	COMMAND
		echo "// This file was automatically generated by 'make' from file '$<'." > cuda_legendre.inc
	COMMAND
		echo "// To modify it, please consider modifying $<" >> cuda_legendre.inc
	COMMAND
		echo "R\"=====(" >> cuda_legendre.inc
	COMMAND
		sed
			-e "s/\\/\\*.*\\*\\///"
			-e "s/\\/\\/.*//"
			-e "/\\/\\*/,/\\*\\//d"
			-e "/^$$/d"
			${CMAKE_CURRENT_SOURCE_DIR}/cuda_legendre.gen.cu
				>> cuda_legendre.inc
	COMMAND
		echo ")=====\"" >> cuda_legendre.inc
	DEPENDS cuda_legendre.gen.cu
	COMMENT "Generating INC codelet cuda_legendre.inc"
	VERBATIM
)

add_custom_target(codelets
	DEPENDS ${CODELET_OUTPUTS} cuda_legendre.inc
)
